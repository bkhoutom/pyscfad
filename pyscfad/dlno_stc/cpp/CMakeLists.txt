cmake_minimum_required(VERSION 3.18)
project(pyscfad_stc_mp2 LANGUAGES CXX)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(Armadillo REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
pybind11_add_module(_stc_mp2 NO_EXTRAS bindings.cpp stc_mp2_kernel.cpp stc_mp2_gradient.cpp stc_mp2_stochastic.cpp stc_mp2_full.cpp)
target_compile_features(_stc_mp2 PRIVATE cxx_std_14)
target_include_directories(_stc_mp2 PRIVATE ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(_stc_mp2 PRIVATE ${ARMADILLO_LIBRARIES} OpenMP::OpenMP_CXX)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_sampling_allocation test_sampling_allocation.cpp)
    target_compile_features(test_sampling_allocation PRIVATE cxx_std_14)
    target_include_directories(test_sampling_allocation PRIVATE ${ARMADILLO_INCLUDE_DIRS})
    target_link_libraries(test_sampling_allocation PRIVATE ${ARMADILLO_LIBRARIES} OpenMP::OpenMP_CXX)
    add_test(NAME sampling_allocation COMMAND test_sampling_allocation)
endif()
